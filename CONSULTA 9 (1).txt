Enunciado: Mostrar el usuario, el tipo de PQRS, el juego relacionado, el pedido relacionado, el administrador que atendió y el estado.

MariaDB [elite_games]> SELECT CONCAT(u.nombre, ' ', u.apellido) AS usuario,
    ->        pq.tipo,
    ->        j.nombre AS juego_relacionado,
    ->        pq.idPedido AS pedido_relacionado,
    ->        CONCAT(a.nombre, ' ', a.apellido) AS administrador,
    ->        pq.estado
    -> FROM pqrs pq
    -> INNER JOIN usuarios u ON pq.idUsuario = u.idUsuario
    -> LEFT JOIN juegos j ON pq.idJuego = j.idJuego
    -> LEFT JOIN administradores a ON pq.idAdmin = a.idAdmin
    -> LEFT JOIN pedidos p ON pq.idPedido = p.idPedido
    -> ORDER BY pq.fechaRegistro DESC;
