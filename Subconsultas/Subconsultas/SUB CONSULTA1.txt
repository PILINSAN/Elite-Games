::Requisito funcional: El sistema debe identificar a los usuarios que han gastado más que el promedio general de gasto de todos los usuarios.
::Enunciado: Mostrar el nombre completo del usuario y su total gastado, solo para quienes superan el promedio general.

MariaDB [elite_games]> SELECT CONCAT(u.nombre, ' ', u.apellido) AS usuario,
    ->        SUM(dp.cantidad * dp.precio_unitario) AS total_gastado
    -> FROM usuarios u
    -> INNER JOIN pedidos p ON u.idUsuario = p.idUsuario
    -> INNER JOIN detalle_pedido dp ON p.idPedido = dp.idPedido
    -> GROUP BY u.idUsuario, u.nombre, u.apellido
    -> HAVING SUM(dp.cantidad * dp.precio_unitario) > (
    ->     SELECT AVG(sub.total)
    ->     FROM (
    ->         SELECT SUM(dp2.cantidad * dp2.precio_unitario) AS total
    ->         FROM pedidos p2
    ->         INNER JOIN detalle_pedido dp2 ON p2.idPedido = dp2.idPedido
    ->         GROUP BY p2.idUsuario
    ->     ) AS sub
    -> );
