Enunciado: Mostrar el nombre completo del usuario, el número de pedido, el nombre del juego, la cantidad y el precio unitario.


MariaDB [elite_games]> SELECT CONCAT(u.nombre, ' ', u.apellido) AS nombre_completo,
    ->        p.idPedido,
    ->        j.nombre AS juego,
    ->        dp.cantidad,
    ->        dp.precio_unitario
    -> FROM usuarios u
    -> INNER JOIN pedidos p ON u.idUsuario = p.idUsuario
    -> INNER JOIN detalle_pedido dp ON p.idPedido = dp.idPedido
    -> INNER JOIN juegos j ON dp.idJuego = j.idJuego
    -> ORDER BY p.idPedido;
