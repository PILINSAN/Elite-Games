Enunciado: Mostrar el usuario, el juego, la categoría, la calificación y la fecha de la reseña (sin hora).

MariaDB [elite_games]> SELECT CONCAT(u.nombre, ' ', u.apellido) AS usuario,
    ->        j.nombre AS juego,
    ->        c.nombre AS categoria,
    ->        r.calificacion,
    ->        DATE(r.fecha_resena) AS fecha_resena
    -> FROM usuarios u
    -> INNER JOIN resenas r ON u.idUsuario = r.idUsuario
    -> INNER JOIN juegos j ON r.idJuego = j.idJuego
    -> INNER JOIN categorias c ON j.idCategoria = c.idCategoria
    -> ORDER BY fecha_resena DESC;
